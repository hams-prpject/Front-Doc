01

JavaScript Introduction

مقدمة تعريفية

ما هي JavaScript؟

JavaScript هي لغة برمجة عالية المستوى، وهي لغة الويب الأساسية إلى جانب HTML و CSS. إذا كانت HTML هي العظام و CSS هي الجلد والملابس، فإن JavaScript هي الدماغ والعضلات التي تجعل الصفحة تتفاعل وتعمل.

  • ▸ تعمل في المتصفح مباشرة بدون تثبيت أي برامج.
  • ▸ تستخدم لتطوير مواقع الويب، تطبيقات الجوال، وحتى الخوادم (Node.js).
  • ▸ ديناميكية، مرنة، ومناسبة للمبتدئين.
  • ▸ هي اللغة رقم 1 الأكثر استخداماً حول العالم.

📝 أول مثال:

hello.js
// أول برنامج لك بـ JavaScript
console.log("مرحباً يا عالم!");
console.log("أهلاً بكم في عالم البرمجة");
— اضغط "تشغيل" لرؤية النتيجة —
🎯 مهمة 1.1:

اكتب برنامج يطبع اسمك، عمرك، وهوايتك المفضلة، كل واحدة في سطر منفصل باستخدام console.log.

JavaScript Syntax

قواعد كتابة الكود

البنية (Syntax) هي مجموعة القواعد التي تحدد كيفية كتابة الكود. أي خطأ بسيط فيها = الكود لن يعمل!

  • ▸ حساسة لحالة الأحرف: name ≠ Name ≠ NAME
  • ▸ تنتهي العبارات بفاصلة منقوطة ; (اختيارية لكن يُنصح بها).
  • ▸ المسافات والأسطر: JavaScript تتجاهل المسافات الزائدة.
  • ▸ النصوص داخل علامات اقتباس: " " أو ' ' أو ` `
syntax.js
// قيم Values: ثوابت ومتغيرات
let x = 10;        // متغير
const PI = 3.14;   // ثابت

// المعرّفات (أسماء المتغيرات) تتبع قواعد:
// - تبدأ بحرف أو _ أو $
// - تتكون من حروف وأرقام
// - حساسة لحالة الأحرف

let firstName = "أحمد";  // ✅ صحيح
let _age = 20;           // ✅ صحيح
let $price = 99;         // ✅ صحيح
// let 1name = "خطأ";    // ❌ لا يبدأ برقم

console.log(firstName, _age, $price);

JavaScript Statements

العبارات والأوامر

العبارة (Statement) هي تعليمة كاملة يقوم المتصفح بتنفيذها. البرنامج عبارة عن سلسلة من العبارات، تنفّذ بالترتيب من الأعلى إلى الأسفل.

كل عبارة تنتهي بفاصلة منقوطة ;، ويمكن أن تتكون من: قيم، عاملين، تعابير، كلمات مفتاحية، تعليقات.

statements.js
// كل سطر هنا = عبارة (statement)
let a = 5;             // عبارة تعريف متغير
let b = 6;             // عبارة تعريف متغير
let c = a + b;         // عبارة حسابية
console.log(c);        // عبارة استدعاء دالة

// يمكن دمج عدة عبارات في سطر واحد بفصلها بـ ;
let x = 1; let y = 2; let z = x + y;
console.log("النتيجة:", z);

// كتلة من العبارات داخل { }
{
  let message = "أنا داخل بلوك";
  console.log(message);
}

المتغيرات — حجر الأساس في البرمجة

ما هو المتغير؟

المتغير عبارة عن صندوق نضع فيه بيانات (نص، رقم، قائمة...) ونعطيه اسماً نستخدمه لاحقاً. تخيّل خزانة صغيرة عليها لاصق باسم age وفي داخلها الرقم 20. كلما احتجت العمر، فتحت الخزانة.

▸ طرق التعريف الثلاث:

🔓

var

الطريقة القديمة (تجنبها).

  • ▸ Function Scope
  • ▸ يمكن إعادة تعريفه
  • ▸ Hoisting بقيمة undefined
✏️

let

للقيم القابلة للتغيير.

  • ▸ Block Scope
  • ▸ لا يمكن إعادة تعريفه
  • ▸ يمكن تغيير قيمته
🔒

const

للقيم الثابتة.

  • ▸ Block Scope
  • ▸ لا يمكن إعادة تعريفه
  • ▸ لا تتغير قيمته
variables.js
// 1) let — قابل للتغيير
let age = 20;
console.log("العمر الآن:", age);
age = 21;                    // ✅ يمكن تغيير القيمة
console.log("العمر بعد سنة:", age);

// 2) const — ثابت
const COUNTRY = "فلسطين";
console.log("الدولة:", COUNTRY);
// COUNTRY = "السعودية";    // ❌ خطأ! لا يمكن تغييره

// 3) إعلانات متعددة
let firstName = "محمد", lastName = "علي", year = 2024;
console.log(firstName, lastName, year);

// 4) متغير بدون قيمة = undefined
let mystery;
console.log("القيمة:", mystery);

JavaScript Data Types

أنواع البيانات

البيانات في JavaScript تنقسم إلى نوعين رئيسيين: بيانات أولية (Primitive) و بيانات مرجعية (Reference).

🟡 الأنواع الأولية (Primitive)

  • String — نص: "مرحبا"
  • Number — رقم: 25, 3.14
  • Boolean — true / false
  • Undefined — متغير لم يُعطَ قيمة
  • Null — قيمة فارغة متعمّدة
  • BigInt — أعداد كبيرة جداً
  • Symbol — قيمة فريدة (متقدم)

🔵 الأنواع المرجعية (Reference)

  • Object — كائن: {name:"أحمد"}
  • Array — مصفوفة: [1, 2, 3]
  • Function — دالة
  • Date — تاريخ
datatypes.js
// String — نص
let name = "أحمد";
let greeting = 'مرحبا';
let template = `أهلاً ${name}`;   // template literal

// Number — رقم (صحيح أو عشري)
let age = 25;
let pi = 3.14;
let big = 1e6;          // 1000000

// Boolean — منطقي
let isStudent = true;
let isOnline = false;

// Undefined — لم يعرف بعد
let unknown;

// Null — فارغ متعمداً
let empty = null;

// Object
let person = { name: "علي", age: 22 };

// Array
let numbers = [10, 20, 30];

// نتعرف على النوع باستخدام typeof
console.log(typeof name);       // string
console.log(typeof age);        // number
console.log(typeof isStudent);  // boolean
console.log(typeof unknown);    // undefined
console.log(typeof person);     // object
console.log(typeof numbers);    // object (المصفوفة نوع من الكائن)

JavaScript Scope

نطاق المتغيرات

النطاق (Scope) يحدد أين يمكن الوصول إلى متغير معين. المتغير الذي تُعرّفه داخل دالة، لا يمكن استخدامه خارجها — إنه محلي داخلها.

ثلاثة أنواع من النطاق:

🌍 Global Scope

معرف خارج أي دالة، ومتاح في كل البرنامج.

🏠 Function Scope

معرف داخل دالة، ومتاح فقط داخلها.

📦 Block Scope

معرف داخل { } (مع let أو const) ومتاح فقط داخلها.

scope.js
// نطاق عام (Global)
let globalVar = "أنا متاح في كل مكان";

function showScope() {
  // نطاق دالة (Function)
  let localVar = "أنا فقط داخل الدالة";
  console.log(globalVar);   // ✅
  console.log(localVar);    // ✅
}

showScope();
console.log(globalVar);     // ✅
// console.log(localVar);   // ❌ خطأ! غير متاح هنا

// نطاق بلوك (Block)
if (true) {
  let blockVar = "أنا داخل البلوك";
  console.log(blockVar);    // ✅
}
// console.log(blockVar);   // ❌ خطأ!

10

JavaScript Operators

العاملين والعمليات

العاملين هي الرموز التي نستخدمها لإجراء العمليات على القيم. تنقسم لعدة أقسام:

١. العاملين الحسابية (Arithmetic):

العامل الوصف مثال النتيجة
+ جمع 5 + 3 8
- طرح 10 - 4 6
* ضرب 4 * 3 12
/ قسمة 10 / 2 5
% باقي القسمة 10 % 3 1
** الأس 2 ** 3 8
++ زيادة 1 x++ x يصبح x+1
-- إنقاص 1 x-- x يصبح x-1
⭐ أهمية إشارة الجمع (+)

إشارة + في JavaScript مميزة جداً وتستخدم لشيئين:

  1. الجمع الحسابي عندما تكون القيم أرقاماً: 5 + 3 = 8
  2. دمج النصوص (Concatenation) عندما تكون إحدى القيمتين نص: "أحمد" + " علي" = "أحمد علي"

انتبه! إذا جمعت رقماً مع نص، JavaScript تحوّل الرقم إلى نص: "5" + 3 = "53" ❌ (وليس 8)

plus-operator.js
// 1) جمع أرقام
console.log(5 + 3);              // 8

// 2) دمج نصوص
console.log("Hello, " + "World"); // "Hello, World"

// 3) الخلط — نتيجة غير متوقعة!
console.log("5" + 3);            // "53"  ← نص!
console.log(5 + "3");            // "53"  ← نص!
console.log(5 + 3 + "5");        // "85"  ← اليسار حسابي ثم نص
console.log("5" + 3 + 5);        // "535" ← الكل نص

// 4) للتحويل العكسي:
console.log(Number("5") + 3);    // 8
console.log(+"5" + 3);           // 8 (الـ + قبل النص يحوّله رقم)
— اضغط "تشغيل" —

٢. عاملين الإسناد (Assignment):

العامل المثال يساوي
= x = 5 x = 5
+= x += 3 x = x + 3
-= x -= 2 x = x - 2
*= x *= 4 x = x * 4
/= x /= 2 x = x / 2
%= x %= 3 x = x % 3

٣. عاملين المقارنة (Comparison):

العامل المعنى مثال
== يساوي (مع تحويل النوع) "5" == 5 → true
=== يساوي بنفس النوع (الأفضل) "5" === 5 → false
!= لا يساوي 5 != 6 → true
!== لا يساوي بنفس النوع 5 !== "5" → true
> أكبر 10 > 5 → true
< أصغر 3 < 5 → true
>= أكبر أو يساوي 5 >= 5 → true
<= أصغر أو يساوي 4 <= 5 → true

٤. العاملين المنطقية (Logical):

العامل المعنى مثال
&& AND — وَ true && false → false
|| OR — أو true || false → true
! NOT — ليس !true → false
🎯 مهام 10:
  1. احسب: (15 + 5) × 2 - 4 / 2 وتأكد من الترتيب.
  2. اكتب برنامجاً يأخذ price و quantity و discount ويحسب الإجمالي.
  3. جرّب: "10" + 5 ثم 10 + "5" ثم 10 + 5 + "5". اشرح الفرق.
  4. قارن: 0 == false و 0 === false. أيّهما أصدق؟

JavaScript Conditions

الجمل الشرطية

الشروط تجعل البرنامج يتخذ قرارات: إذا تحقق شرط، نفّذ شيء، وإلّا نفّذ شيئاً آخر.

if / else if / else

conditions.js
let grade = 75;

if (grade >= 90) {
  console.log("ممتاز 🌟");
} else if (grade >= 80) {
  console.log("جيد جداً 👏");
} else if (grade >= 70) {
  console.log("جيد 👍");
} else if (grade >= 60) {
  console.log("مقبول");
} else {
  console.log("راسب 😞");
}

// عامل ثلاثي (Ternary): شرط ? نتيجة_True : نتيجة_False
let age = 18;
let status = age >= 18 ? "بالغ" : "قاصر";
console.log("الحالة:", status);

JavaScript Loops

الحلقات التكرارية

الحلقات تكرر تنفيذ كتلة من الكود طالما تحقق شرط معيّن. توفّر لنا الجهد والوقت.

١. for — عندما نعرف عدد التكرارات

for-loop.js
// طباعة الأرقام من 1 إلى 5
for (let i = 1; i <= 5; i++) {
  console.log("العدد:", i);
}

// جدول ضرب الـ 3
console.log("--- جدول الـ 3 ---");
for (let i = 1; i <= 10; i++) {
  console.log(`3 × ${i} = ${3 * i}`);
}

// مجموع الأعداد من 1 إلى 100
let sum = 0;
for (let i = 1; i <= 100; i++) {
  sum += i;
}
console.log("المجموع:", sum);
— اضغط "تشغيل" —

٢. while — طالما تحقق الشرط

while-loop.js
let i = 1;
while (i <= 5) {
  console.log("التكرار رقم", i);
  i++;
}

// do...while: ينفّذ مرة واحدة على الأقل
let n = 10;
do {
  console.log("القيمة:", n);
  n++;
} while (n < 13);
— اضغط "تشغيل" —

٣. for...of — للعناصر في مصفوفة

for-of.js
let fruits = ["تفاح", "موز", "برتقال", "عنب"];

// for...of: للحصول على القيم
for (let fruit of fruits) {
  console.log("🍎", fruit);
}

// for...in: للحصول على المفاتيح/الفهارس
for (let i in fruits) {
  console.log(`الفهرس ${i}: ${fruits[i]}`);
}
— اضغط "تشغيل" —
⚠️ break و continue
  • ▸ break: يكسر الحلقة ويخرج منها فوراً.
  • ▸ continue: يتجاوز التكرار الحالي ويذهب للذي بعده.

الدوال — لُبّ البرمجة الحديثة

ما هي الدالة؟

الدالة هي كتلة من الكود نعطيها اسماً، ونستدعيها كلما احتجناها. تساعدنا على إعادة استخدام الكود بدلاً من تكراره، وتقسيم البرنامج إلى أجزاء صغيرة سهلة الفهم.

تتكون الدالة من: اسم + معاملات (Parameters) + جسم + (اختياري) قيمة مرجَعة.

١. الدالة العادية (Function Declaration)

functions-1.js
// 1) دالة بدون معاملات وبدون قيمة مرجَعة
function sayHello() {
  console.log("مرحباً 👋");
}
sayHello();   // استدعاء

// 2) دالة بمعاملات
function greet(name) {
  console.log("مرحباً يا " + name);
}
greet("أحمد");
greet("ليلى");

// 3) دالة ترجع قيمة (return)
function add(a, b) {
  return a + b;
}
let result = add(5, 3);
console.log("الناتج:", result);   // 8

// 4) قيم افتراضية للمعاملات
function welcome(name = "زائر") {
  return `أهلاً يا ${name}`;
}
console.log(welcome());           // أهلاً يا زائر
console.log(welcome("سارة"));    // أهلاً يا سارة

٣. Arrow Functions (السهمية) — الأشهر حالياً

arrow-functions.js
// الصيغة: (المعاملات) => { الجسم }

// كاملة
const square = (n) => {
  return n * n;
};
console.log(square(7));         // 49

// مختصرة (return تلقائي)
const cube = n => n * n * n;
console.log(cube(3));           // 27

// مع عدة معاملات
const sum = (a, b, c) => a + b + c;
console.log(sum(1, 2, 3));      // 6

// بدون معاملات
const sayHi = () => console.log("Hi!");
sayHi();

JavaScript Objects

الكائنات

الكائن مجموعة من الأزواج (مفتاح: قيمة). يستخدم لوصف عنصر له خصائص متعددة: شخص، سيارة، منتج، إلخ.

objects.js
// إنشاء كائن
const student = {
  firstName: "أحمد",
  lastName: "علي",
  age: 20,
  isActive: true,
  hobbies: ["قراءة", "كرة قدم"],
  greet: function() {
    return `أهلاً، أنا ${this.firstName}`;
  }
};

// الوصول للخصائص
console.log(student.firstName);      // الطريقة 1: النقطة
console.log(student["lastName"]);    // الطريقة 2: الأقواس
console.log(student.age);
console.log(student.greet());        // استدعاء الدالة الداخلية

// التعديل والإضافة
student.age = 21;                    // تعديل
student.email = "ahmed@mail.com";    // إضافة جديدة
console.log(student);

// الحذف
delete student.isActive;
console.log(student);

// التكرار على الخصائص
for (let key in student) {
  console.log(`${key}: ${student[key]}`);
}

array-methods.js
let nums = [1, 2, 3, 4, 5];

// map: ضرب كل عنصر في 2
const doubled = nums.map(n => n * 2);
console.log(doubled);                  // [2, 4, 6, 8, 10]

// filter: العناصر الزوجية فقط
const evens = nums.filter(n => n % 2 === 0);
console.log(evens);                    // [2, 4]

// reduce: مجموع العناصر
const sum = nums.reduce((acc, n) => acc + n, 0);
console.log("المجموع:", sum);          // 15

// find: أول عنصر أكبر من 3
const first = nums.find(n => n > 3);
console.log(first);                    // 4

// forEach: تنفيذ دالة لكل عنصر
nums.forEach((n, i) => console.log(`الفهرس ${i}: ${n}`));

// sort + reverse
const sorted = [3, 1, 4, 1, 5, 9, 2].sort();
console.log(sorted);                   // [1, 1, 2, 3, 4, 5, 9]
console.log(sorted.reverse());         // [9, 5, 4, 3, 2, 1, 1]

// join
console.log(nums.join(" + "));         // "1 + 2 + 3 + 4 + 5"
